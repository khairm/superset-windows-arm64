```ts
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { toast } from "@superset/ui/sonner";
import { getHostServiceClientByUrl } from "renderer/lib/host-service-client";
import { useWorkspaceHostUrl } from "renderer/hooks/host-service/useWorkspaceHostUrl";
// or: import { useHostUrl } from "renderer/hooks/host-service/useHostTargetUrl";
// or: import { useLocalHostService } from "renderer/routes/_authenticated/providers/LocalHostServiceProvider";

export function useThing(workspaceId: string) {
	const hostUrl = useWorkspaceHostUrl(workspaceId); // string | null
	const queryClient = useQueryClient();

	const query = useQuery({
		queryKey: ["host-service", "thing", hostUrl, workspaceId] as const,
		enabled: Boolean(hostUrl),
		queryFn: () =&gt; {
			if (!hostUrl) throw new Error("Host unavailable");
			return getHostServiceClientByUrl(hostUrl).someRouter.get.query({ workspaceId });
		},
	});

	const mutation = useMutation({
		mutationFn: (input: { foo: string }) =&gt; {
			if (!hostUrl) throw new Error("Host unavailable");
			return getHostServiceClientByUrl(hostUrl).someRouter.set.mutate({ workspaceId, ...input });
		},
		onSuccess: () =&gt; queryClient.invalidateQueries({ queryKey: ["host-service", "thing", hostUrl, workspaceId] }),
		onError: (err) =&gt; toast.error(err instanceof Error ? err.message : String(err)),
	});

	return { query, mutation };
}
```

Gotchas that will bite you:
- `getHostServiceClientByUrl` takes a **URL**, not a hostId or port. `getHostServiceClient(port)` is the loopback shortcut.

- Always include `hostUrl` (or `machineId`) in the query key — otherwise two hosts share a cache entry.
- Set `networkMode: "always"` on anything that must work offline against `127.0.0.1` (the global default already does this; keep it if you override defaultOptions locally).
- New procedures: `.query` → `queryProcedure` (5s default timeout, raise with `.meta({ timeoutMs })`); `.mutation` → `protectedProcedure`.</result>
